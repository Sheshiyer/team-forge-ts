    /// Status alias table — mirrors teamforge-vault-parity.mjs:281-290.
    fn normalize_status(raw: &str, tags: &[String]) -> String {
        if tags.iter().any(|t| t.eq_ignore_ascii_case("archived")) {
            return "archived".to_string();
        }
        match raw.trim().to_ascii_lowercase().as_str() {
            "in progress" | "in-progress" | "active" => "active".to_string(),
            "cancelled" | "canceled" | "completed" => "completed".to_string(),
            "paused" | "on hold" | "on-hold" => "paused".to_string(),
            "draft" => "draft".to_string(),
            "planning" => "planning".to_string(),
            "white-labelable" | "white labelable" => "white-labelable".to_string(),
            other => other.to_string(),
        }
    }

    /// Three-source workspace_id fallback — mirrors teamforge-vault-parity.mjs:864-878.
    fn resolve_workspace_id(frontmatter_value: Option<&str>, arg_value: Option<&str>) -> Option<String> {
        if let Some(v) = frontmatter_value.and_then(|s| (!s.is_empty()).then(|| s.to_string())) { return Some(v); }
        if let Some(v) = arg_value.and_then(|s| (!s.is_empty()).then(|| s.to_string())) { return Some(v); }
        std::env::var("TEAMFORGE_WORKSPACE_ID").ok().filter(|s| !s.is_empty())
    }

    /// `external_refs` decoder — mirrors the regex at :169-178.
    /// Input shape from gray_matter: Vec<Value> where each Value is a map { "system": ..., "id": ... }.
    fn decode_external_refs(refs: &[Value]) -> Vec<(String, String)> {
        refs.iter().filter_map(|r| {
            let system = r.get("system").and_then(Value::as_str)?;
            let id = r.get("id").and_then(|v| v.as_str().map(String::from).or_else(|| v.as_i64().map(|n| n.to_string())))?;
            Some((system.to_string(), id))
        }).collect()
    }

    /// mergeArtifacts dedup — mirrors :1310-1333.
    fn merge_artifacts(artifacts: Vec<ArtifactRecord>) -> Vec<ArtifactRecord> {
        let mut seen: BTreeMap<String, ArtifactRecord> = BTreeMap::new();
        for a in artifacts {
            let key = match (&a.source, &a.external_id) {
                (s, Some(id)) => format!("{s}:{id}"),
                (s, None) => format!("{s}::{}", a.title.as_deref().unwrap_or("")),
            };
            seen.entry(key).or_insert(a);
        }
        seen.into_values().collect()
    }

    /// resolveEmployeeForKpi — mirrors :959-994.
    /// Returns Some(employee_id) on exact or prefix match, None on unresolved.
    fn resolve_employee_for_kpi(
        member_id: &str,
        roster: &HashMap<String, String>, // alias -> canonical employee_id
    ) -> Option<String> {
        // Tier 1: exact alias match (case-insensitive).
        let lc = member_id.to_ascii_lowercase();
        if let Some(emp) = roster.get(&lc) { return Some(emp.clone()); }
        // Tier 2: prefix match — any alias that starts with member_id.
        for (alias, emp) in roster {
            if alias.starts_with(&lc) { return Some(emp.clone()); }
        }
        // Tier 3: unresolved.
        None
    }
